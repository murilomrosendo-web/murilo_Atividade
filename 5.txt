using System;

class Program
{
    static int Fatorial(int n)
    {
        int resultado = 1;

        for (int i = 1; i <= n; i++)
        {
            resultado *= i;
        }

        return resultado;
    }

    static void Main()
    {
        Console.Write("Digite um número: ");
        int num = int.Parse(Console.ReadLine());

        Console.WriteLine("Fatorial: " + Fatorial(num));
    }
}
